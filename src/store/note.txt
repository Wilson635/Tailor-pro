// STATISTIQUES CALCULÉES
// ==========================================

const computeStatistics = (
    clients: Client[],
    orders: Order[]
): Statistics => {
  const now = new Date();
  const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);

  const monthlyOrders = orders.filter(
      (o) => new Date(o.createdAt) >= startOfMonth
  );
  const monthlyRevenue = monthlyOrders.reduce((s, o) => s + o.advancePayment, 0);

  const prevMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);
  const prevMonthOrders = orders.filter((o) => {
    const d = new Date(o.createdAt);
    return d >= prevMonthStart && d < startOfMonth;
  });
  const prevRevenue = prevMonthOrders.reduce((s, o) => s + o.advancePayment, 0);
  const revenueGrowth =
      prevRevenue > 0
          ? Math.round(((monthlyRevenue - prevRevenue) / prevRevenue) * 100)
          : 0;

  const unpaidOrders = orders.filter((o) => o.paymentStatus !== 'paid');

  return {
    totalClients: clients.length,
    monthlyRevenue,
    revenueGrowth,
    ordersInProgress: orders.filter((o) => o.orderStatus === 'in_progress').length,
    unpaidInvoices: unpaidOrders.length,
    unpaidAmount: unpaidOrders.reduce((s, o) => s + o.remainingAmount, 0),
    completedOrders: orders.filter((o) => o.orderStatus === 'completed').length,
    totalExpenses: 0, // À implémenter si nécessaire
    netProfit: monthlyRevenue,
  };
};

export const useAppStore = create<AppState>((set, get) => ({
  // ── État initial ──
  isAuthenticated: false,
  userId: null,

  clients: [],
  orders: [],
  catalog: [],
  activities: [],
  statistics: {
    totalClients: 0,
    monthlyRevenue: 0,
    revenueGrowth: 0,
    ordersInProgress: 0,
    unpaidInvoices: 0,
    unpaidAmount: 0,
    completedOrders: 0,
    totalExpenses: 0,
    netProfit: 0,
  },
  measurements: {},
  payments: {},

  isLoading: false,
  error: null,
  selectedClientFilter: 'all',
  selectedCatalogCategory: 'all',
  searchQuery: '',

  // ── Auth ──
  setAuthenticated: (status, userId) =>
      set({ isAuthenticated: status, userId: userId ?? null }),

  logout: () =>
      set({
        isAuthenticated: false,
        userId: null,
        clients: [],
        orders: [],
        catalog: [],
        activities: [],
        measurements: {},
        payments: {},
      }),

  // ── Chargement Firebase ──
  loadClients: async () => {
    const { userId } = get();
    if (!userId) return;
    const clients = await FirestoreService.fetchClients(userId);
    const statistics = computeStatistics(clients, get().orders);
    set({ clients, statistics });
  },

  loadOrders: async () => {
    const { userId } = get();
    if (!userId) return;
    const orders = await FirestoreService.fetchOrders(userId);
    const statistics = computeStatistics(get().clients, orders);
    set({ orders, statistics });
  },

  loadCatalog: async () => {
    const { userId } = get();
    if (!userId) return;
    const catalog = await FirestoreService.fetchCatalog(userId);
    set({ catalog });
  },

  loadActivities: async () => {
    const { userId } = get();
    if (!userId) return;
    const activities = await FirestoreService.fetchActivities(userId);
    set({ activities });
  },

  loadAll: async () => {
    const { userId } = get();
    if (!userId) return;
    set({ isLoading: true, error: null });
    try {
      const [clients, orders, catalog, activities] = await Promise.all([
        FirestoreService.fetchClients(userId),
        FirestoreService.fetchOrders(userId),
        FirestoreService.fetchCatalog(userId),
        FirestoreService.fetchActivities(userId),
      ]);
      const statistics = computeStatistics(clients, orders);
      set({ clients, orders, catalog, activities, statistics });
    } catch (e: any) {
      set({ error: e.message ?? 'Erreur de chargement' });
    } finally {
      set({ isLoading: false });
    }
  },

  // ── Clients ──
  addClient: async (data) => {
    const { userId } = get();
    if (!userId) return;
    const client = await FirestoreService.createClient(userId, data);
    const clients = [client, ...get().clients];
    set({ clients, statistics: computeStatistics(clients, get().orders) });
    // Log activité
    await FirestoreService.logActivity(userId, {
      type: 'new_client',
      title: 'Nouveau client',
      subtitle: data.fullName,
      timestamp: new Date(),
      clientId: client.id,
    });
  },

  updateClient: async (clientId, data) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.updateClient(userId, clientId, data);
    const clients = get().clients.map((c) =>
        c.id === clientId ? { ...c, ...data, updatedAt: new Date() } : c
    );
    set({ clients });
  },

  deleteClient: async (clientId) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.deleteClient(userId, clientId);
    const clients = get().clients.filter((c) => c.id !== clientId);
    set({ clients, statistics: computeStatistics(clients, get().orders) });
  },

  // ── Orders ──
  addOrder: async (data) => {
    const { userId } = get();
    if (!userId) return;
    const order = await FirestoreService.createOrder(userId, data);
    const orders = [order, ...get().orders];
    set({ orders, statistics: computeStatistics(get().clients, orders) });
    await FirestoreService.logActivity(userId, {
      type: 'new_order',
      title: 'Nouvelle commande',
      subtitle: data.clientName,
      timestamp: new Date(),
      clientId: data.clientId,
      orderId: order.id,
    });
  },

  updateOrder: async (orderId, data) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.updateOrder(userId, orderId, data);
    const orders = get().orders.map((o) =>
        o.id === orderId ? { ...o, ...data, updatedAt: new Date() } : o
    );
    set({ orders, statistics: computeStatistics(get().clients, orders) });
  },

  deleteOrder: async (orderId) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.deleteOrder(userId, orderId);
    const orders = get().orders.filter((o) => o.id !== orderId);
    set({ orders, statistics: computeStatistics(get().clients, orders) });
  },

  // ── Measurements ──
  saveMeasurements: async (clientId, data) => {
    const { userId } = get();
    if (!userId) return;
    const measurements = await FirestoreService.saveMeasurements(userId, clientId, data);
    set((state) => ({
      measurements: { ...state.measurements, [clientId]: measurements },
    }));
  },

  loadMeasurements: async (clientId) => {
    const { userId } = get();
    if (!userId) return;
    const measurements = await FirestoreService.fetchMeasurements(userId, clientId);
    if (measurements) {
      set((state) => ({
        measurements: { ...state.measurements, [clientId]: measurements },
      }));
    }
  },

  // ── Payments ──
  addPayment: async (orderId, data) => {
    const { userId } = get();
    if (!userId) return;
    const payment = await FirestoreService.addPayment(userId, orderId, data);
    set((state) => ({
      payments: {
        ...state.payments,
        [orderId]: [...(state.payments[orderId] ?? []), payment],
      },
    }));
    // Recharger les commandes pour mettre à jour les statuts
    await get().loadOrders();
    // Log activité
    await FirestoreService.logActivity(userId, {
      type: 'payment_received',
      title: 'Paiement reçu',
      subtitle: data.clientId,
      amount: data.amount,
      timestamp: new Date(),
      clientId: data.clientId,
      orderId,
    });
  },

  loadPayments: async (orderId) => {
    const { userId } = get();
    if (!userId) return;
    const payments = await FirestoreService.fetchPayments(userId, orderId);
    set((state) => ({
      payments: { ...state.payments, [orderId]: payments },
    }));
  },

  // ── Catalog ──
  addCatalogModel: async (data) => {
    const { userId } = get();
    if (!userId) return;
    const model = await FirestoreService.createCatalogModel(userId, data);
    set((state) => ({ catalog: [model, ...state.catalog] }));
  },

  updateCatalogModel: async (modelId, data) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.updateCatalogModel(userId, modelId, data);
    set((state) => ({
      catalog: state.catalog.map((m) =>
          m.id === modelId ? { ...m, ...data } : m
      ),
    }));
  },

  deleteCatalogModel: async (modelId) => {
    const { userId } = get();
    if (!userId) return;
    await FirestoreService.deleteCatalogModel(userId, modelId);
    set((state) => ({
      catalog: state.catalog.filter((m) => m.id !== modelId),
    }));
  },

  toggleCatalogFavorite: async (modelId) => {
    const model = get().catalog.find((m) => m.id === modelId);
    if (!model) return;
    await get().updateCatalogModel(modelId, { isFavorite: !model.isFavorite });
  },

  // ── UI ──
  setLoading: (loading) => set({ isLoading: loading }),
  setError: (error) => set({ error }),
  setClientFilter: (filter) => set({ selectedClientFilter: filter }),
  setCatalogCategory: (category) => set({ selectedCatalogCategory: category }),
  setSearchQuery: (query) => set({ searchQuery: query }),

  // ── Helpers synchrones ──
  getClientById: (clientId) =>
      get().clients.find((c) => c.id === clientId),
  getOrdersByClient: (clientId) =>
      get().orders.filter((o) => o.clientId === clientId),
  getMeasurementsByClient: (clientId) =>
      get().measurements[clientId],
  getPaymentsByOrder: (orderId) =>
      get().payments[orderId] ?? [],
}));
